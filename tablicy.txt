create table books(
    id serial primary key,
    title varchar(32),
    author varchar(32),
    genre varchar(16)
);

create table loans(
    id serial primary key,
    book_id int,
    borrower varchar(32),
    date_issued date,
    date_due date
    foreign key(book_id) references books(id)
);
